SLOT 6
Run it

Cat fact page

from flask import Flask, render_template
import requests

app = Flask(__name__)

@app.route("/")
def home():
    r = requests.get(
        "https://catfact.ninja/fact",
        timeout=5)
    fact = r.json()["fact"]
    return render_template("index.html",
                           fact=fact)
