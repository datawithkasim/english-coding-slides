SLOT 6
Run it

Checkout session

import stripe, os
stripe.api_key = os.environ["STRIPE_TEST_KEY"]

@app.route("/buy", methods=["POST"])
def buy():
    s = stripe.checkout.Session.create(
        line_items=[{
          "price_data": {
            "currency": "krw",
            "product_data": {"name": "Cookie"},
            "unit_amount": 2000},
          "quantity": 1}],
        mode="payment",
        success_url="http://localhost:5000/ok",
        cancel_url="http://localhost:5000/")
    return redirect(s.url)
