Auto-jump bot

import time
from selenium import webdriver
from selenium.webdriver.common.keys import Keys

d = webdriver.Chrome()
d.get("chrome://dino")
body = d.find_element("tag name", "body")
body.send_keys(Keys.SPACE)

while True:
    body.send_keys(Keys.SPACE)
    time.sleep(0.6)

기본 봇 — 0.6초마다 점프
