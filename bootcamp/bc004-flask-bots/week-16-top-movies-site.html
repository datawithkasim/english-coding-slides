Run it

Top 10 page

@app.route("/")
def home():
    movies = (Movie.query
        .order_by(Movie.rating.desc())
        .limit(10).all())
    return render_template("top.html", movies=movies)

@app.route("/add", methods=["GET","POST"])
def add():
    form = MovieForm()
    if form.validate_on_submit():
        db.session.add(Movie(
            title=form.title.data,
            rating=form.rating.data))
        db.session.commit()
        return redirect("/")
    return render_template("add.html", form=form)

⚠️ Watch out

Return template instead of redirect

✗ broken 
db.session.commit()
return render_template("add.html")

→ refresh re-submits!

✓ fixed 
db.session.commit()
return redirect("/")

→ post/redirect/get
