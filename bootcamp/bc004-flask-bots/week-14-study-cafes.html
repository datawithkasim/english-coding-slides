SLOT 6
Run it

Add + show cafes

@app.route("/add", methods=["GET", "POST"])
def add():
    form = CafeForm()
    if form.validate_on_submit():
        with open("cafes.csv", "a", newline="") as f:
            csv.writer(f).writerow([
                form.name.data, form.wifi.data,
                form.power.data, form.quiet.data,
            ])
        return redirect("/list")
    return render_template("add.html", form=form)

🐛 Debugactivity

Escape commas

csv.writer(f).writerow([name, ...])

Reveal fix

The csv module quotes for you. Don't write the comma yourself — use writerow.
