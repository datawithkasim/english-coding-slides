SLOT 6
Run it

Cheapest ICN → NRT

import requests, os
url    = "https://tequila-api.kiwi.com/v2/search"
key    = os.getenv("TEQUILA_KEY")
params = {
    "fly_from": "ICN", "fly_to": "NRT",
    "date_from": "01/07/2026",
    "date_to":   "15/07/2026",
    "curr": "KRW", "limit": 1,
}
data = requests.get(url, params=params, headers={"apikey":key}).json()
flight = data["data"][0]
print(f"{flight['price']} KRW")
▸ Output
320000 KRW
