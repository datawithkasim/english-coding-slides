from tkinter import messagebox

def save():
    site = site_e.get()
    user = user_e.get()
    pw   = pw_e.get()

    if not site or not pw:
        messagebox.showerror("Oops", "Empty fields")
        return

    with open("pw.txt", "a") as f:
        f.write(f"{site} | {user} | {pw}\n")

SLOT 9
⚠️ Watch out

File overwritten

✗ broken 
with open("pw.txt", "w") as f:
    f.write(line)

"w" wipes old data.

✓ fixed 
with open("pw.txt", "a") as f:
    f.write(line)

"a" keeps history.
