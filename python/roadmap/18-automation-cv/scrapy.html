Code

Example

import scrapy

class QuoteSpider(scrapy.Spider):
    name = 'quotes'
    start_urls = ['http://quotes.toscrape.com']

    def parse(self, response):
        for q in response.css('div.quote'):
            yield {'text': q.css('span.text::text').get()}
