Code

Example

import customtkinter as ctk
ctk.set_appearance_mode('dark')
app = ctk.CTk()
ctk.CTkLabel(app, text='Hello').pack(pady=10)
ctk.CTkButton(app, text='Go').pack()
app.mainloop()
