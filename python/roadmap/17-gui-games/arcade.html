Code

Example

import arcade

class Game(arcade.Window):
    def __init__(self):
        super().__init__(640, 480, 'Game')
    def on_draw(self):
        self.clear()
        arcade.draw_circle_filled(100, 100, 30, arcade.color.RED)

Game()
arcade.run()
