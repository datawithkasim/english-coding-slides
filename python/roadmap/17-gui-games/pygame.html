Code

Example

import pygame
pygame.init()
screen = pygame.display.set_mode((640, 480))
running = True
while running:
    for e in pygame.event.get():
        if e.type == pygame.QUIT: running = False
    screen.fill((0, 0, 0))
    pygame.display.flip()
