from langchain_anthropic import ChatAnthropic
from langchain.prompts import ChatPromptTemplate

llm = ChatAnthropic(model='claude-3-5-sonnet')
prompt = ChatPromptTemplate.from_messages([('user', '{q}')])
chain = prompt | llm
chain.invoke({'q': 'hi'})

The bug

    print("Adult")
