Functions

One per scene piece

def draw_ground(): ...
def draw_turret(x): ...
def draw_marker(x): ...
def draw_stars(): ...

def handle_keys(gun_x):
    keys = pygame.key.get_pressed()
    if keys[pygame.K_LEFT] and gun_x > MIN_X:
        gun_x -= GUN_SPEED
    if keys[pygame.K_RIGHT] and gun_x < MAX_X:
        gun_x += GUN_SPEED
    return gun_x
Game loop

Now beautifully tidy

gun_x = WIDTH // 2
running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

    gun_x = handle_keys(gun_x)

    screen.fill(SKY)
    draw_stars()
    draw_ground()
    draw_turret(gun_x)
    draw_marker(gun_x)
    pygame.display.flip()
    clock.tick(60)
