SLOT 6
Run it

Action menu

actions = ["fight", "run", "talk"]
action = input("What do you do? ").lower().strip()

while action not in actions:
    action = input("Pick fight, run, or talk: ").lower().strip()

if action == "fight": print("⚔️ Swing!")
elif action == "run":   print("🏃 Escape.")
else:                  print("🤝 New friend?")
