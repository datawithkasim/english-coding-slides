def handle_keys(ship_x, ship_y):
    keys = pygame.key.get_pressed()
    if keys[pygame.K_LEFT]:  ship_x -= SPEED
    if keys[pygame.K_RIGHT]: ship_x += SPEED
    if keys[pygame.K_UP]:    ship_y -= SPEED
    if keys[pygame.K_DOWN]:  ship_y += SPEED
    ship_x = max(20, min(WIDTH - 20, ship_x))
    ship_y = max(20, min(HEIGHT - 60, ship_y))
    return ship_x, ship_y
Game loop

Now beautifully tidy

ship_x, ship_y = WIDTH // 2, HEIGHT - 100
running = True

while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_SPACE:
                bullets.append([ship_x, ship_y - 20])

    ship_x, ship_y = handle_keys(ship_x, ship_y)

    screen.fill(SPACE)
    draw_stars()
    for e in enemies: draw_enemy(e)
    for b in bullets: draw_bullet(b)
    draw_ship(ship_x, ship_y)
    pygame.display.flip()
    clock.tick(60)

⚠ Edge cases

Gotchas + pitfalls

Frame rate matters
Without a clock, the game runs as fast as the CPU allows — chaos.
Coordinates start top-left
y increases as you go DOWN, not up.
Always handle QUIT
Forget pygame.event.QUIT and your window won't close.
Convert images
img.convert() or .convert_alpha() — way faster than not.
