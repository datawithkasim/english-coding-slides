shop.py — finished

money = 100
spent = 0
items = 0
print(f"You have {money} dollars")
while money > 0:
    price = int(input("Price? "))
    if price > money:
        print("Too expensive!")
        break
    money -= price
    spent += price
    items += 1
    print(f"You have {money} dollars")
print("Shop closed!")
money += 50
print(f"Now you have {money} dollars")
print(f"You spent {spent} dollars")
print(f"You bought {items} things")

money = 100
spent = 0
items = 0
basket = []
print(f"You have {money} dollars")
while money > 0:
    item = input("What is it? ")
    if item == "stop":
        break
    price = float(input("Price? "))
    if price > money:
        print("Too expensive!")
    else:
        money -= price
        spent += price
        items += 1
        basket.append(item)
        print(f"You have {money} dollars")
print("Shop closed!")
money += 50
print(f"Now you have {money} dollars")
print(f"You spent {spent} dollars")
print(f"You bought {len(basket)} things")
for thing in basket:
    print(f"- {thing}")
if items > 0:
    print(f"Average: {round(spent / items, 2)}")
