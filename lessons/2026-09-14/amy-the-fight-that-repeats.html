hp = 100
while hp > 0:
    move = ""
    while move not in ["attack", "block"]:
        move = input("attack or block? ").lower()
    if move == "attack":
        damage = 20
    else:
        damage = 5
    hp = hp - damage
    hearts = int(hp / 20)
    print("♥" * hearts)
    print("HP:", hp)
    if hp > 60:
        print("You are fine.")
    elif hp > 20:
        print("Careful!")
    else:
        print("Almost dead!")
