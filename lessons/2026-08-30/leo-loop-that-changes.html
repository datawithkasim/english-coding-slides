def on_pillars():
    x = 2
    spacing = 2
    for i in range(5):
        blocks.fill(QUARTZ_BLOCK,
            pos(x, 0, 0),
            pos(x, 4, 0), REPLACE)
        x = x + spacing
        spacing = spacing + 1

player.on_chat("pillars", on_pillars)
