def rainbow():
    x = 2
    spacing = 2
    colors = [RED_WOOL, ORANGE_WOOL, YELLOW_WOOL,
        LIME_WOOL, LIGHT_BLUE_WOOL, PURPLE_WOOL]
    for i in range(30):
        y = 2 + i % 6
        blocks.fill(colors[i % 6],
            pos(x, 0, 0),
            pos(x, y, 0), REPLACE)
        if i % 3 == 0:
            blocks.fill(GOLD_BLOCK,
                pos(x, y + 1, 0),
                pos(x, y + 1, 0), REPLACE)
        x = x + spacing

player.on_chat("rainbow", rainbow)
