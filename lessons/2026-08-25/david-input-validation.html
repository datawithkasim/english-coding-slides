v4 — check every line

inventory = ["torch"]
valid_choices = ["strike", "leave"]

strike_or_leave = input("Strike or leave? ").lower()

while strike_or_leave not in valid_choices:
    print("Type strike or leave.")
    strike_or_leave = input("Strike or leave? ").lower()

if strike_or_leave == "strike":
    print("The rock splits. A diamond!")
    inventory.append("diamond")
elif strike_or_leave == "leave":
    if "torch" in inventory:
        print("Your torch lights the way out.")
    else:
        print("You stumble in the dark...")
