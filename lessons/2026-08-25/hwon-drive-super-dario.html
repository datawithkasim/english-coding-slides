# --- INITIALIZATION ---
pygame.init()

WIDTH, HEIGHT = 400, 400
screen = pygame.display.set_mode((WIDTH, HEIGHT))
pygame.display.set_caption("Super Dario!!!")
clock = pygame.time.Clock()
font = pygame.font.Font(pygame.font.get_default_font(), 14)

# --- COLORS --- (your ten colour lines stay exactly as they are)
BACKGROUND_COLOR = (135, 205, 225)

super_dario = {
    'x': 20,   'y': 100,
    'h': 40,   'w': 40,
    'color': RED,  'speed': 6,
}
bread = {
    'x': 320,  'y': 300,
    'h': 20,   'w': 15,

}

Full program · the loop
running = True
while running:
    for event in pygame.event.get():
        if event.type == pygame.QUIT:
            running = False

    keys = pygame.key.get_pressed()
    if keys[pygame.K_RIGHT]:
        super_dario['x'] += super_dario['speed']
    if keys[pygame.K_LEFT]:
        super_dario['x'] -= super_dario['speed']
    if keys[pygame.K_UP]:
        super_dario['y'] -= super_dario['speed']
    if keys[pygame.K_DOWN]:
        super_dario['y'] += super_dario['speed']

    if super_dario['x'] < 0: super_dario['x'] = 0
    if super_dario['x'] > WIDTH - super_dario['w']: super_dario['x'] = WIDTH - super_dario['w']
    if super_dario['y'] < 0: super_dario['y'] = 0
    if super_dario['y'] > HEIGHT - super_dario['h']: super_dario['y'] = HEIGHT - super_dario['h']

    # -------- 3. RENDERING LAYER --------
    screen.fill(BACKGROUND_COLOR)
    pygame.draw.rect(screen, RED, (super_dario['x'], super_dario['y'], super_dario['w'], super_dario['h']))
    pygame.draw.rect(screen, PINK, (bread['x'], bread['y'], bread['w'], bread['h']))
    pygame.display.flip()
    clock.tick(60)

pygame.quit()
