Full program
def on_stairs():
    y = 0
    rows = 5
    for layer in range(5):
        z = 0
        for row in range(rows):
            blocks.fill(STONE,
                pos(0, y, z),
                pos(7, y, z), REPLACE)
            z = z + 1
        y = y + 1
        rows = rows - 1

player.on_chat("stairs", on_stairs)
