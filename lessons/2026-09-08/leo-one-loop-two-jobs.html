def rainbow():
    x = 2
    spacing = 2
    colors = [RED_WOOL, ORANGE_WOOL, YELLOW_WOOL,
        LIME_WOOL, LIGHT_BLUE_WOOL, PURPLE_WOOL]
    for i in range(12):
        if i % 3 == 0:
            height = 9
        else:
            height = 3
        blocks.fill(colors[i % 6],
            pos(x, 0, 0),
            pos(x, height, 0), REPLACE)
        if i % 2 == 0:
            blocks.fill(STONE,
                pos(x, height + 1, 0),
                pos(x, height + 1, 0), REPLACE)
        else:
            blocks.fill(GOLD_BLOCK,
                pos(x, height + 1, 0),
                pos(x, height + 1, 0), REPLACE)
        x = x + spacing
