    print("HP:", hp)
    if hp > 60:

import random

hp = 100
turn = 0
monster_hp = 60
while hp > 0 and monster_hp > 0:
    turn += 1
    print("--- Turn", turn, "---")
    move = ""
    while move not in ["attack", "block", "run"]:
        move = input("attack or block? ").lower()
    if move == "run":
        break
    if move == "attack":
        damage = random.randint(10, 25)
        monster_hp = monster_hp - 20
    else:
        damage = 5
    hp -= damage
    print("Monster:", monster_hp)
    hearts = int(hp / 20)
    print("♥" * hearts)
    print("HP:", hp)
    if hp > 60:
        print("You are fine.")
    elif hp > 20:
        print("Careful!")
    else:
        print("Almost dead!")

if monster_hp <= 0:
    print("You win!")
else:
    print("You lose!")
