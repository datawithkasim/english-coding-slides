LANDING
🎯 The program

shop.py — finished

money = 100
spent = 0
items = 0
print(f"You have {money} dollars")
while money > 0:
    price = int(input("Price? "))
    if price > money:
        print("Too expensive!")
        break
    money -= price
    spent += price
    items += 1
    print(f"You have {money} dollars")
print("Shop closed!")
money += 50
print(f"Now you have {money} dollars")
print(f"You spent {spent} dollars")
print(f"You bought {items} things")
print("Thanks for shopping!")  # your SOLO 2 line
