def make_tree(x, z, leaf=CHERRY_LEAVES):
    blocks.fill(OAK_LOG, pos(x, 0, z), pos(x, 4, z))
    blocks.fill(leaf,
        pos(x - 1, 5, z - 1), pos(x + 1, 6, z + 1))
    return x + 10

def make_house(x, z):
    blocks.fill(OAK_PLANKS,
        pos(x, 0, z), pos(x + 4, 3, z + 4))
    blocks.place(GLASS, pos(x + 2, 2, z))

def on_forest():
    leaves_type = [CHERRY_LEAVES, DARK_OAK_LEAVES,
        PALE_OAK_LEAVES, OAK_LEAVES]
    x = 1
    for i in range(4):
        x = make_tree(x, 1, leaves_type[i])
    make_house(1, 11)

player.on_chat("forest", on_forest)
