    x_end = 2;  y_end = 3;  z_end = 2

    leaf_x_end = 3;  leaf_y_end = 6;  leaf_z_end = 3

        blocks.fill(OAK_LOG,
            pos(x_start, y_start, z_start),

            pos(leaf_x_start, leaf_y_start, leaf_z_start),

        z_start = z_start + 4;  z_end = z_end + 4
        leaf_y_start = leaf_y_start + 1;  leaf_y_end = leaf_y_end + 1
        leaf_z_start = leaf_z_start + 4;  leaf_z_end = leaf_z_end + 4
