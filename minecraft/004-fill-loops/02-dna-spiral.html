XS = [6, 4, 0, -4, -6, -4, 0, 4]
ZS = [0, 4, 6, 4, 0, -4, -6, -4]

def on_chat():
    for i in range(0, 24):
        k = i % 8
        ax = 8 + XS[k]
        az = 8 + ZS[k]
        j = (k + 4) % 8
        bx = 8 + XS[j]
        bz = 8 + ZS[j]
        blocks.fill(GOLD_BLOCK,
            pos(ax, i, az), pos(ax + 1, i, az + 1), REPLACE)
        blocks.fill(EMERALD_BLOCK,
            pos(bx, i, bz), pos(bx + 1, i, bz + 1), REPLACE)
        if i % 4 == 0:
            blocks.fill(QUARTZ_BLOCK,
                pos(ax, i, az),
                pos(bx, i, bz), REPLACE)
player.on_chat("dna", on_chat)
