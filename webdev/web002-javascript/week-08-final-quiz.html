Goal

What it does 🎯

  • 5+ multiple-choice questions
  • Show one at a time
  • +1 score for right answer
  • Final score after last question
  • Restart button
Concept

Question = an object

let q = {
  q: "Capital of Korea?",
  choices: ["Tokyo", "Seoul", "Beijing"],
  answer: 1
};

Object = labeled values. Like a tiny dictionary.

키-값 묶음

Concept

Array of question objects

let questions = [
  { q: "Capital of Korea?",
    choices: ["Tokyo", "Seoul", "Beijing"],
    answer: 1 },
  { q: "2 + 2 = ?",
    choices: ["3", "4", "5"],
    answer: 1 }
];

Logic

Show one question

let current = 0;
let score = 0;

function showQuestion() {
  let q = questions[current];
  document.querySelector("#q").textContent = q.q;
  document.querySelector("#opt1").textContent = q.choices[0];
  document.querySelector("#opt2").textContent = q.choices[1];
  document.querySelector("#opt3").textContent = q.choices[2];
}
Logic

Check answer

function check(chosen) {
  if (chosen === questions[current].answer) {
    score = score + 1;
  }
  current = current + 1;
  if (current < questions.length) {
    showQuestion();
  } else {
    document.querySelector("#q").textContent = "Done!";
  }
  document.querySelector("#score").textContent = "Score: " + score;
}
Wire it up

Connect buttons

document.querySelector("#opt1").addEventListener("click", () => check(0));
document.querySelector("#opt2").addEventListener("click", () => check(1));
document.querySelector("#opt3").addEventListener("click", () => check(2));

showQuestion();  // start!
